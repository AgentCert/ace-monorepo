```graphql
# ApplicationSpec — Full Catalog Types (spec §17.2)

"""
Full application spec as read from catalog/apps/<tier>/<name>/app.yaml.
This is the richer catalog type; AppHubEntry is the simpler deployment-status type.
"""
type ApplicationSpec {
  name:               String!
  displayName:        String!
  version:            String!
  tier:               String!             # "official" | "community"
  domain:             String!
  capabilityDomains:  [String!]!
  tags:               [String!]!
  description:        CatalogAppDescription!
  install:            CatalogInstallSpec!
  healthProbe:        CatalogHealthProbeSpec!
  loadTest:           CatalogLoadTestSpec!
  microservices:      [CatalogMicroserviceSpec!]!
  faultCompatibility: [FaultCompatibilityEntry!]!
  inputs:             [CatalogAppInput!]!
  schemaVersion:      String!
}

type CatalogAppDescription {
  short:          String!
  long:           String!
  suitableFor:    [String!]!
  notSuitableFor: [String!]!
}

type CatalogInstallSpec {
  method:     String!
  folder:     String
  chartRef:   CatalogChartRef
  namespace:  CatalogNamespaceSpec!
  timeout:    String!
  wait:       Boolean!
}

type CatalogChartRef {
  repo:    String!
  chart:   String!
  version: String!
}

type CatalogNamespaceSpec {
  default:      String!
  configurable: Boolean!
}

type CatalogHealthProbeSpec {
  urlTemplate:         String!
  expectedStatus:      String!
  initialDelaySeconds: Int!
  periodSeconds:       Int!
  failureThreshold:    Int!
}

type CatalogLoadTestSpec {
  enabled: Boolean!
  method:  String
  image:   String
  args:    [String!]
}

type CatalogMicroserviceSpec {
  name:          String!
  displayName:   String!
  description:   String
  k8sLabel:      String!
  k8sKind:       String!
  k8sNamespace:  String!
  criticality:   String!
  relevantFaults: [String!]!
  dependsOn:     [String!]!
}

type FaultCompatibilityEntry {
  faultName:          String!
  compatible:         Boolean!
  notes:              String
  recommendedTargets: [String!]!
}

type CatalogAppInput {
  key:         String!
  displayName: String!
  description: String
  type:        String!
  required:    Boolean!
  default:     String
  helmPath:    String!
  values:      [String!]
  min:         Int
  max:         Int
  unit:        String
  advanced:    Boolean!
}

# ──────────────────────────────────────────────────
# New Catalog Queries (additive — existing queries unchanged)
# ──────────────────────────────────────────────────

extend type Query {
  """List all applications in the catalog (official first, then community; alphabetical within tier)"""
  listApplications(projectID: ID!): [ApplicationSpec!]! @authorized

  """Get a single application by name"""
  getApplication(projectID: ID!, appName: String!): ApplicationSpec @authorized
}
```

**Note:** Type names are prefixed with `Catalog` (e.g., `CatalogInstallSpec`) to avoid collision with the existing `InstallSpec` or similar types in `models_gen.go`.

### Task 3: Create `pkg/catalog/mapper.go`

**File to Create:** `AgentCert/chaoscenter/graphql/server/pkg/catalog/mapper.go`

```go
package catalog

import (
	"github.com/litmuschaos/litmus/chaoscenter/graphql/server/graph/model"
)

// toGraphQLModel maps an internal AppCatalogEntry to the gqlgen-generated model.ApplicationSpec.
// Implements the placeholder in service.go.
func toGraphQLModel(e *AppCatalogEntry) *model.ApplicationSpec {
	if e == nil {
		return nil
	}

	spec := &model.ApplicationSpec{
		Name:              e.Metadata.Name,
		DisplayName:       e.Metadata.DisplayName,
		Version:           e.Metadata.Version,
		Tier:              e.Metadata.Tier,
		Domain:            e.Metadata.Domain,
		CapabilityDomains: e.Metadata.CapabilityDomains,
		Tags:              e.Metadata.Tags,
		SchemaVersion:     "1",
		Description: &model.CatalogAppDescription{
			Short:          e.Spec.Description.Short,
			Long:           e.Spec.Description.Long,
			SuitableFor:    e.Spec.Description.SuitableFor,
			NotSuitableFor: e.Spec.Description.NotSuitableFor,
		},
		Install: &model.CatalogInstallSpec{
			Method:  e.Spec.Install.Method,
			Folder:  strPtr(e.Spec.Install.Folder),
			Timeout: e.Spec.Install.Timeout,
			Wait:    e.Spec.Install.Wait,
			Namespace: &model.CatalogNamespaceSpec{
				Default:      e.Spec.Install.Namespace.Default,
				Configurable: e.Spec.Install.Namespace.Configurable,
			},
		},
		HealthProbe: &model.CatalogHealthProbeSpec{
			URLTemplate:         e.Spec.HealthProbe.URL,
			ExpectedStatus:      e.Spec.HealthProbe.ExpectedStatus,
			InitialDelaySeconds: defaultInt(e.Spec.HealthProbe.InitialDelaySeconds, 30),
			PeriodSeconds:       defaultInt(e.Spec.HealthProbe.PeriodSeconds, 10),
			FailureThreshold:    defaultInt(e.Spec.HealthProbe.FailureThreshold, 6),
		},
		LoadTest: &model.CatalogLoadTestSpec{
			Enabled: e.Spec.LoadTest.Enabled,
			Method:  strPtr(e.Spec.LoadTest.Method),
			Image:   strPtr(e.Spec.LoadTest.Image),
			Args:    e.Spec.LoadTest.Args,
		},
	}

	// ChartRef (external-helm only)
	if e.Spec.Install.ChartRef != nil {
		spec.Install.ChartRef = &model.CatalogChartRef{
			Repo:    e.Spec.Install.ChartRef.Repo,
			Chart:   e.Spec.Install.ChartRef.Chart,
			Version: e.Spec.Install.ChartRef.Version,
		}
	}

	// Microservices
	for _, ms := range e.Spec.Microservices {
		ns := ms.K8s.Namespace
		if ns == "" {
			ns = "{{.AppNamespace}}"
		}
		spec.Microservices = append(spec.Microservices, &model.CatalogMicroserviceSpec{
			Name:          ms.Name,
			DisplayName:   ms.DisplayName,
			Description:   strPtr(ms.Description),
			K8sLabel:      ms.K8s.Label,
			K8sKind:       ms.K8s.Kind,
			K8sNamespace:  ns,
			Criticality:   defaultStr(ms.Criticality, "medium"),
			RelevantFaults: ms.RelevantFaults,
			DependsOn:     ms.DependsOn,
		})
	}

	// FaultCompatibility
	for _, fc := range e.Spec.FaultCompatibility {
		spec.FaultCompatibility = append(spec.FaultCompatibility, &model.FaultCompatibilityEntry{
			FaultName:          fc.FaultName,
			Compatible:         fc.Compatible,
			Notes:              strPtr(fc.Notes),
			RecommendedTargets: fc.RecommendedTargets,
		})
	}

	// Inputs
	for _, inp := range e.Spec.Inputs {
		spec.Inputs = append(spec.Inputs, &model.CatalogAppInput{
			Key:         inp.Key,
			DisplayName: inp.DisplayName,
			Description: strPtr(inp.Description),
			Type:        inp.Type,
			Required:    inp.Required,
			Default:     strPtr(inp.Default),
			HelmPath:    inp.HelmPath,
			Values:      inp.Values,
			Min:         inp.Min,
			Max:         inp.Max,
			Unit:        strPtr(inp.Unit),
			Advanced:    inp.Advanced,
		})
	}

	// Ensure non-nil slices
	if spec.Microservices == nil {
		spec.Microservices = []*model.CatalogMicroserviceSpec{}
	}
	if spec.FaultCompatibility == nil {
		spec.FaultCompatibility = []*model.FaultCompatibilityEntry{}
	}
	if spec.Inputs == nil {
		spec.Inputs = []*model.CatalogAppInput{}
	}
	if spec.Tags == nil {
		spec.Tags = []string{}
	}
	if spec.CapabilityDomains == nil {
		spec.CapabilityDomains = []string{}
	}

	return spec
}

func strPtr(s string) *string {
	if s == "" {
		return nil
	}
	return &s
}

func defaultStr(s, def string) string {
	if s == "" {
		return def
	}
	return s
}

func defaultInt(v, def int) int {
	if v == 0 {
		return def
	}
	return v
}
```

```go
// ListApplications is the resolver for listApplications.
func (r *queryResolver) ListApplications(ctx context.Context, projectID string) ([]*model.ApplicationSpec, error) {
    if r.catalogService == nil {
        return []*model.ApplicationSpec{}, nil
    }
    return r.catalogService.ListApplications(ctx, projectID)
}

// GetApplication is the resolver for getApplication.
func (r *queryResolver) GetApplication(ctx context.Context, projectID string, appName string) (*model.ApplicationSpec, error) {
    if r.catalogService == nil {
        return nil, nil
    }
    return r.catalogService.GetApplication(ctx, projectID, appName)
}
```

## Common Issues and Solutions

### Issue: Type name collision with existing types
**Symptom:** `gqlgen: type "ChartRef" already defined`  
**Solution:** Use `Catalog`-prefixed names for all new types (`CatalogChartRef`, `CatalogInstallSpec`, etc.) as specified in Task 1.

### Issue: resolver.go interface not satisfied after code gen
**Symptom:** `does not implement QueryResolver: missing method ListApplications`  
**Solution:** Run `go generate ./...` first, THEN add the resolver methods. The generated interface must exist before the resolver can implement it.

### Issue: `catalogService` is nil in resolver
**Symptom:** catalog queries return empty but no error  
**Solution:** Check `CATALOG_DIR` env var is set and points to a directory containing `apps/official/sock-shop/app.yaml`. The nil-guard in the resolver returns `[]` on nil service — intentional.
